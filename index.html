// ============================================================ // UI Elements & Menus — IMADCRAFT 3D (Chunk Edition) // ============================================================ const mainMenu = document.getElementById("main-menu"); const settingsMenu = document.getElementById("settings-menu"); const pauseMenu = document.getElementById("pause-menu"); const menuStart = document.getElementById("menu-start"); const menuSettings = document.getElementById("menu-settings"); const settingsBack = document.getElementById("settings-back"); const pauseResume = document.getElementById("pause-resume"); const pauseSettings = document.getElementById("pause-settings"); const pauseSave = document.getElementById("pause-save"); const pauseExit = document.getElementById("pause-exit"); const btnPause = document.getElementById("btn-pause"); const btnSave = document.getElementById("btn-save"); const btnCamera = document.getElementById("btn-camera"); let settingsReturnTo = "main"; let hasUnsavedChanges = false; // ============================================================ // 🎮 بدء اللعبة (مع تحميل العالم وتوليد الـ Chunks) // ============================================================ menuStart.addEventListener("click", () => { mainMenu.classList.add("hidden"); settingsMenu.classList.add("hidden"); pauseMenu.classList.add("hidden"); isGameStarted = true; // 1) محاولة تحميل حفظ سابق const loaded = loadWorld(); // 2) توليد الـ Chunks حول اللاعب const pcx = Math.floor(player.pos.x / CHUNK_SIZE); const pcz = Math.floor(player.pos.z / CHUNK_SIZE); for (let dx = -1; dx <= 1; dx++) { for (let dz = -1; dz <= 1; dz++) { forceGenerateChunk(pcx + dx, pcz + dz); } } requestChunksAround(player.pos.x, player.pos.z); // 3) ضبط ارتفاع اللاعب فوق الأرض إن كان عالياً if (!loaded) { const bx = Math.floor(player.pos.x); const bz = Math.floor(player.pos.z); let y = WORLD_HEIGHT - 1; while (y > 0 && !isSolid(getBlockAt(bx, y, bz))) y--; player.pos.y = y + 1.05; player.vel.set(0, 0, 0); } // 4) طلب Pointer Lock (سطح المكتب فقط) if (!isTouchDevice()) { try { renderer.domElement.requestPointerLock(); } catch (e) {} } showToast(loaded ? "✅ تم تحميل العالم" : "🌍 عالم جديد"); }); // ============================================================ // ⚙️ إعدادات القائمة الرئيسية // ============================================================ menuSettings.addEventListener("click", () => { settingsReturnTo = "main"; mainMenu.classList.add("hidden"); pauseMenu.classList.add("hidden"); settingsMenu.classList.remove("hidden"); }); // ============================================================ // ↩️ الرجوع من الإعدادات // ============================================================ settingsBack.addEventListener("click", () => { settingsMenu.classList.add("hidden"); if (settingsReturnTo === "pause" && isGameStarted) { pauseMenu.classList.remove("hidden"); } else { mainMenu.classList.remove("hidden"); } }); // ============================================================ // ⏸ إيقاف مؤقت (مع فك Pointer Lock) // ============================================================ btnPause.addEventListener("click", () => { if (!isGameStarted) { showToast("ابدأ اللعبة أولاً!"); return; } // تصفير الحركة لمنع "الانزلاق" عند الاستئناف moveState.forward = 0; moveState.backward = 0; moveState.left = 0; moveState.right = 0; moveState.jump = false; // فك Pointer Lock حتى يظهر المؤشر if (document.pointerLockElement) document.exitPointerLock(); pauseMenu.classList.remove("hidden"); }); // ============================================================ // ▶️ استئناف اللعب (مع إعادة Pointer Lock) // ============================================================ pauseResume.addEventListener("click", () => { pauseMenu.classList.add("hidden"); if (!isTouchDevice()) { try { renderer.domElement.requestPointerLock(); } catch (e) {} } }); // ============================================================ // ⚙️ الإعدادات من قائمة الإيقاف // ============================================================ pauseSettings.addEventListener("click", () => { settingsReturnTo = "pause"; pauseMenu.classList.add("hidden"); settingsMenu.classList.remove("hidden"); }); // ============================================================ // 💾 حفظ من قائمة الإيقاف // ============================================================ if (pauseSave) { pauseSave.addEventListener("click", () => { saveWorld(); hasUnsavedChanges = false; }); } // ============================================================ // 🏠 العودة للقائمة الرئيسية // ============================================================ pauseExit.addEventListener("click", () => { // تحذير إن كان هناك تعديلات غير محفوظة if (hasUnsavedChanges) { if (!confirm("لديك تعديلات غير محفوظة. هل تريد الخروج بدون حفظ؟")) { return; } } pauseMenu.classList.add("hidden"); settingsMenu.classList.add("hidden"); mainMenu.classList.remove("hidden"); isGameStarted = false; // تصفير الحركة moveState.forward = 0; moveState.backward = 0; moveState.left = 0; moveState.right = 0; moveState.jump = false; // فك Pointer Lock if (document.pointerLockElement) document.exitPointerLock(); }); // ============================================================ // 💾 حفظ العالم (متوافق مع نظام Chunk) // ============================================================ btnSave.addEventListener("click", () => { saveWorld(); hasUnsavedChanges = false; }); // ============================================================ // 👤 تبديل المنظور (يعمل فعلياً مع الكاميرا) // ============================================================ btnCamera.addEventListener("click", () => { cameraMode = (cameraMode + 1) % 2; btnCamera.textContent = cameraMode === 0 ? "👤 منظور 1" : "👤 منظور 3"; showToast(cameraMode === 0 ? "كاميرا: الشخص الأول" : "كاميرا: الشخص الثالث"); }); // ============================================================ // ⌨️ اختصار لوحة المفاتيح — Escape // ============================================================ document.addEventListener("keydown", (e) => { if (e.code !== "Escape") return; if (!isGameStarted) return; if (!pauseMenu.classList.contains("hidden")) { pauseResume.click(); } else if (!settingsMenu.classList.contains("hidden")) { settingsBack.click(); } else if (pauseMenu.classList.contains("hidden") && mainMenu.classList.contains("hidden")) { btnPause.click(); } }); // ============================================================ // 🔄 الإيقاف التلقائي عند مغادرة التبويب // ============================================================ document.addEventListener("visibilitychange", () => { if (document.hidden && isGameStarted) { // تصفير الحركة moveState.forward = 0; moveState.backward = 0; moveState.left = 0; moveState.right = 0; moveState.jump = false; if (pauseMenu.classList.contains("hidden") && settingsMenu.classList.contains("hidden")) { btnPause.click(); } } }); // ============================================================ // ⚠️ تحذير عند إغلاق الصفحة بتعديلات غير محفوظة // ============================================================ window.addEventListener("beforeunload", (e) => { if (hasUnsavedChanges) { e.preventDefault(); e.returnValue = ""; } }); // ============================================================ // 🔗 دوال مساعدة يجب توفرها من الكود الأساسي: // - isTouchDevice() // كشف الموبايل // - loadWorld() // من قسم الحفظ // - saveWorld() // من قسم الحفظ // - forceGenerateChunk() // من قسم Chunk // - requestChunksAround() // من قسم Chunk // - getBlockAt() // من قسم World Access // - isSolid() // من قسم World Access // - showToast() // من قسم Utils // - player, moveState, yaw, pitch, cameraMode, // isGameStarted, CHUNK_SIZE, WORLD_HEIGHT, // renderer, scene, camera // متغيرات عامة // ============================================================
